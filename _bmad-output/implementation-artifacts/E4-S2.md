## 📋 STORY CONTEXT

### Objetivo
Implementar lógica backend para identificar "leads en riesgo" (que no han avanzado de estado en 7+ días) y exponer widget en frontend con lista de leads en riesgo + contador.

### Problema que Resuelve
- Ejecutivos no saben qué leads necesitan seguimiento urgente
- Leads olvidados sin progresión (venden menos)
- AC-MVP-R3 & Aging Report requirement: "Leads atrasados"

### Decisión de Planning Review
> **Definición de "En Riesgo":** Any lead que no ha cambiado de estado en 7+ días (aplica a TODOS los estados: Nuevo, En contacto, Propuesta).

### User Story
> **Como** Ejecutivo de Ventas  
> **Quiero** ver una lista de "Leads en Riesgo" (sin progresión en 7 días)  
> **Para que** pueda priorizar follow-ups a leads atrasados

### Backend Endpoint
```python
# FastAPI endpoint: GET /api/leads/at-risk

@router.get("/api/leads/at-risk")
async def get_leads_at_risk(db: Session = Depends(get_db)):
    """
    Returns list of leads that haven't changed status in 7+ days.
    Ordered by days_without_change DESC (oldest first).
    """
    
    # Define "at risk" threshold
    RISK_THRESHOLD_DAYS = 7
    risk_threshold = datetime.now() - timedelta(days=RISK_THRESHOLD_DAYS)
    
    # Query: leads where last_status_change_at < 7 days ago
    # Exclude "Cerrado" status (completed leads)
    at_risk_leads = (
        db.query(Lead)
        .filter(
            Lead.last_status_change_at <= risk_threshold,
            Lead.status != LeadStatus.CERRADO
        )
        .order_by(Lead.last_status_change_at.asc())  # Oldest first
        .all()
    )
    
    # Calculate days_without_change for each lead
    response = []
    for lead in at_risk_leads:
        days_without_change = (datetime.now() - lead.last_status_change_at).days
        response.append({
            "id": lead.id,
            "name": lead.name,
            "company": lead.company,
            "email": lead.email,
            "status": lead.status.value,
            "priority": lead.priority.value,
            "days_without_change": days_without_change,
            "created_at": lead.created_at,
            "last_status_change_at": lead.last_status_change_at
        })
    
    return {"data": response, "count": len(response)}
```

// components/LeadsAtRiskWidget.tsx

interface LeadsAtRiskWidget {
  leads: Lead[];
  loading: boolean;
  expanded: boolean;
}

export function LeadsAtRiskWidget() {
  const [atRiskLeads, setAtRiskLeads] = useState<Lead[]>([]);
  const [isExpanded, setIsExpanded] = useState(false);
  const [loading, setLoading] = useState(false);
  
  // Fetch on mount + auto-refresh every 5 minutes
  useEffect(() => {
    fetchAtRiskLeads();
    const interval = setInterval(fetchAtRiskLeads, 5 * 60 * 1000);
    return () => clearInterval(interval);
  }, []);
  
  const fetchAtRiskLeads = async () => {
    setLoading(true);
    try {
      const response = await fetch('/api/leads/at-risk');
      const data = await response.json();
      setAtRiskLeads(data.data);
    } finally {
      setLoading(false);
    }
  };
  
  const count = atRiskLeads.length;
  const isAtRisk = count > 0;
  
  return (
    <div className={`widget ${isAtRisk ? 'bg-red-50' : 'bg-green-50'}`}>
      {loading ? (
        <Spinner />
      ) : (
        <>
          <button onClick={() => setIsExpanded(!isExpanded)}>
            {isAtRisk ? (
              <span className="text-red-600">⚠️ {count} Leads en Riesgo</span>
            ) : (
              <span className="text-green-600">✅ Todos en día</span>
            )}
          </button>
          
          {isExpanded && isAtRisk && (
            <LeadsAtRiskPanel leads={atRiskLeads} onLeadClick={handleLeadClick} />
          )}
        </>
      )}
    </div>
  );
}

// Panel showing list of at-risk leads
function LeadsAtRiskPanel({ leads, onLeadClick }) {
  return (

            key={lead.id}
            onClick={() => onLeadClick(lead)}
            className="p-2 hover:bg-red-50 cursor-pointer border-l-2 border-red-300"
          >
            <div className="font-semibold">{lead.name}</div>
            <div className="text-sm text-gray-600">
              {lead.company} • {lead.status} • <span className="text-red-600 font-bold">{lead.days_without_change} días</span>
            </div>
          </li>
        ))}
      </ul>
    </div>
  );
}
